cmake_minimum_required(VERSION 3.16)
project(usb-player-core-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt5 Qt6 COMPONENTS Core REQUIRED)
find_package(Qt${QT_VERSION_MAJOR} COMPONENTS Core REQUIRED)
add_subdirectory(../../../recipes-common/common/files common)
include(../files/src/PlayerCore.cmake)
add_executable(usb-player-core-test playlist_test.cpp ../files/qml/qml.qrc)
set_target_properties(usb-player-core-test PROPERTIES AUTORCC ON)
target_link_libraries(usb-player-core-test PRIVATE player-core)
target_compile_options(player-core PRIVATE -Wall -Wextra -Werror)
target_compile_options(usb-player-core-test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME usb-playlist-and-player COMMAND usb-player-core-test)
set_tests_properties(usb-playlist-and-player PROPERTIES TIMEOUT 20)
