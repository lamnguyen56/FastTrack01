cmake_minimum_required(VERSION 3.16)
project(usb-player-core-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt5 Qt6 COMPONENTS Core Qml REQUIRED)
find_package(Qt${QT_VERSION_MAJOR} COMPONENTS Core Qml REQUIRED)
add_subdirectory(../../../recipes-common/common/files common)
include(../files/src/PlayerCore.cmake)
add_executable(usb-player-core-test playlist_test.cpp ../files/qml/qml.qrc)
set_target_properties(usb-player-core-test PROPERTIES AUTORCC ON)
target_link_libraries(usb-player-core-test PRIVATE player-core)
target_compile_options(player-core PRIVATE -Wall -Wextra -Werror)
target_compile_options(usb-player-core-test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME usb-playlist-and-player COMMAND usb-player-core-test)
set_tests_properties(usb-playlist-and-player PROPERTIES TIMEOUT 20)

add_library(player-qml-types STATIC ../files/src/Ui/QmlTypes.cpp)
target_link_libraries(player-qml-types PUBLIC player-core Qt${QT_VERSION_MAJOR}::Qml)

add_executable(usb-player-transport-test transport_test.cpp ../files/qml/qml.qrc)
set_target_properties(usb-player-transport-test PROPERTIES AUTOMOC ON AUTORCC ON)
target_link_libraries(usb-player-transport-test PRIVATE player-qml-types)
target_compile_options(usb-player-transport-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME usb-player-async-transport COMMAND usb-player-transport-test)
set_tests_properties(usb-player-async-transport PROPERTIES TIMEOUT 20)
